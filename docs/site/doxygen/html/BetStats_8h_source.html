69//----------------------------------------------------------------
70//
71// Selects which recorded outcomes BetStats::tally() adds together.
72//
73// ComeOut : decisions reached on a come out roll (stored with no pivot)
74// Points : decisions reached after a point/pivot was established
75// (PassLine4..10, Come4..10, ...)
76// All : ComeOut + Points
77//
78enum class BetScope { ComeOut, Points, All };
79
80//----------------------------------------------------------------
81//
82// How a roll dependent bet fared on a particular number.
83//
84// EntireLoss : the whole wager lost (e.g. World on a point number)
85// PartialWin : net win, but some components lost (Horn on 2, C&E on 11)
86// BreakEven : a component won but net profit is zero (World on 7)
87// CleanWin : wins with no losing components (Field, AnySeven, ...)
88//
89enum class RollOutcome { EntireLoss, PartialWin, BreakEven, CleanWin };
90
91// Hardway losses split by what ended the bet.
92struct HardwayLosses
93{
94 BetStat easy; // Rolled the number the easy way
95 BetStat seven; // Rolled a seven
96};
