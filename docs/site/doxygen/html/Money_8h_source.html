28// Enum for formatting flags
29enum class MoneyFormat : uint8_t
30{
31 ShowDollar = 1 << 0,
32 ShowCents = 1 << 1,
33 ShowComma = 1 << 2
34};
35
36// Per-stream storage for flags
37inline long& getMoneyFormat(std::ios_base& ios)
38{
39 static int index = std::ios_base::xalloc();
40 return ios.iword(index); // returns long&
41}
42
43// Manipulators
44inline std::ios_base& showDollar(std::ios_base& os)
45{
46 getMoneyFormat(os) |= static_cast<long>(MoneyFormat::ShowDollar);
47 return os;
48}
49
50inline std::ios_base& noDollar(std::ios_base& os)
